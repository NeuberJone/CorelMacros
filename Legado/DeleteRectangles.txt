    Dim shp As Shape
    Dim subShp As Shape
    Dim i As Integer, j As Integer

    ' Reference the active page
    Set currentPage = ActiveDocument.ActivePage

    For i = currentPage.Shapes.Count To 1 Step -1
        Set shp = currentPage.Shapes(i)

        ' Delete directly if the shape is a rectangle
        If shp.Type = cdrRectangleShape Then
            shp.Delete
